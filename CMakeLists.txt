cmake_minimum_required(VERSION 3.16)
project(hk_camera_driver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(Threads REQUIRED)

set(MVS_ROOT "/opt/MVS" CACHE PATH "MVS SDK root")
find_path(MVS_INCLUDE_DIR NAMES MvCameraControl.h HINTS "${MVS_ROOT}/include")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(MVS_LIBRARY_HINT "${MVS_ROOT}/lib/aarch64")
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(MVS_LIBRARY_HINT "${MVS_ROOT}/lib/64")
else()
  set(MVS_LIBRARY_HINT "${MVS_ROOT}/lib/32")
endif()
find_library(MVS_LIBRARY NAMES MvCameraControl HINTS "${MVS_LIBRARY_HINT}" "${MVS_ROOT}/lib")
if(NOT MVS_INCLUDE_DIR OR NOT MVS_LIBRARY)
  message(FATAL_ERROR "Install MVS SDK; set MVS_ROOT or MVS_INCLUDE_DIR and MVS_LIBRARY to actual SDK paths")
endif()
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${MVS_INCLUDE_DIR}")
set(CMAKE_REQUIRED_LIBRARIES "${MVS_LIBRARY}")
check_cxx_source_compiles("#include <MvCameraControl.h>\nint main(){MV_CC_Initialize(); MV_CC_Finalize();return 0;}" HK_MVS_HAS_INITIALIZE)
unset(CMAKE_REQUIRED_INCLUDES)
unset(CMAKE_REQUIRED_LIBRARIES)
message(STATUS "MVS headers: ${MVS_INCLUDE_DIR}; library: ${MVS_LIBRARY}; global initialization: ${HK_MVS_HAS_INITIALIZE}")

add_executable(hk_camera_node src/hk_camera_node.cpp)
target_include_directories(hk_camera_node PRIVATE include "${MVS_INCLUDE_DIR}" ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(hk_camera_node rclcpp rcl_interfaces sensor_msgs std_msgs std_srvs)
target_link_libraries(hk_camera_node "${MVS_LIBRARY}" ${OpenCV_LIBS} Threads::Threads)
if(HK_MVS_HAS_INITIALIZE)
  target_compile_definitions(hk_camera_node PRIVATE HK_MVS_HAS_INITIALIZE)
endif()
get_filename_component(MVS_LIBRARY_DIR "${MVS_LIBRARY}" DIRECTORY)
set_target_properties(hk_camera_node PROPERTIES INSTALL_RPATH "${MVS_LIBRARY_DIR}")
add_executable(image_rate_probe src/image_rate_probe.cpp)
ament_target_dependencies(image_rate_probe rclcpp sensor_msgs)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(hk_camera_node PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(image_rate_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS hk_camera_node image_rate_probe DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test tests/core_test.cpp)
  target_include_directories(core_test PRIVATE include)
  target_link_libraries(core_test Threads::Threads)
  add_test(NAME driver_core COMMAND core_test)
endif()
ament_package()
